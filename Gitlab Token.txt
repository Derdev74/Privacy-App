Personal Data Deletion Automation: Feasibility Study & Implementation Guide
Executive Summary
A privacy-first, local-only automation tool for requesting data deletion from online services. Core principle: All data processing happens on your machine—no third-party exposure.
Verdict: Highly feasible for email accounts, impractical for phone numbers.

import keyring
import json
from pathlib import Path

class TokenManager:
    def __init__(self):
        self.service_name = "DataDeletionTool"
        
    def save_token(self, account_email, token_data):
        """Securely store OAuth tokens in OS keychain"""
        keyring.set_password(
            self.service_name, 
            account_email, 
            json.dumps(token_data)
        )
        
    def get_token(self, account_email):
        """Retrieve OAuth tokens from OS keychain"""
        token_json = keyring.get_password(self.service_name, account_email)
        return json.loads(token_json) if token_json else None
        
    def delete_token(self, account_email):
        """Remove tokens from keychain"""
        keyring.delete_password(self.service_name, account_email)

from google.auth.transport.requests import Request
from google.oauth2.credentials import Credentials
from google_auth_oauthlib.flow import Flow
from googleapiclient.discovery import build
import pickle

class GmailAuthenticator:
    SCOPES = ['https://www.googleapis.com/auth/gmail.readonly']
    
    def __init__(self, credentials_file='config/google_credentials.json'):
        self.credentials_file = credentials_file
        self.token_manager = TokenManager()
        
    def authenticate(self, email_address):
        """Complete OAuth flow and return authenticated service"""
        # Check for existing token
        token_data = self.token_manager.get_token(email_address)
        
        creds = None
        if token_data:
            creds = Credentials.from_authorized_user_info(token_data, self.SCOPES)
            
        # Refresh or get new token
        if not creds or not creds.valid:
            if creds and creds.expired and creds.refresh_token:
                creds.refresh(Request())
            else:
                flow = Flow.from_client_secrets_file(
                    self.credentials_file, 
                    self.SCOPES,
                    redirect_uri='urn:ietf:wg:oauth:2.0:oob'
                )
                auth_url, _ = flow.authorization_url(prompt='consent')
                
                print(f"Open this URL in your browser:\n{auth_url}")
                code = input("Enter the authorization code: ")
                flow.fetch_token(code=code)
                creds = flow.credentials
                
            # Save the credentials
            self.token_manager.save_token(email_address, creds.to_json())
            
        return build('gmail', 'v1', credentials=creds)

from datetime import datetime
import re

class EmailScanner:
    def __init__(self, gmail_service):
        self.service = gmail_service
        
    def discover_companies(self, max_results=500):
        """Scan emails for service registrations"""
        # Optimized search query
        query = '''
            from:(-me) 
            ("welcome to" OR "verify your email" OR "account created" 
             OR "subscription confirmed" OR "registration successful")
            after:2015/1/1
        '''
        
        companies = set()
        page_token = None
        
        while len(companies) < max_results:
            # Fetch only headers, not body
            results = self.service.users().messages().list(
                userId='me',
                q=query,
                pageToken=page_token,
                maxResults=min(100, max_results - len(companies))
            ).execute()
            
            messages = results.get('messages', [])
            if not messages:
                break
                
            # Batch fetch headers only
            for msg in messages:
                msg_data = self.service.users().messages().get(
                    userId='me',
                    id=msg['id'],
                    format='metadata',
                    metadataHeaders=['From']
                ).execute()
                
                # Extract domain from sender
                sender = msg_data['payload']['headers'][0]['value']
                domain = self.extract_domain(sender)
                if domain:
                    companies.add(domain)
                    
            page_token = results.get('nextPageToken')
            if not page_token:
                break
                
        return list(companies)
    
    def extract_domain(self, sender_string):
        """Extract domain from email header"""
        # Pattern: something@domain.com
        match = re.search(r'@([a-zA-Z0-9.-]+\.[a-zA-Z]{2,})', sender_string)
        if match:
            domain = match.group(1).lower()
            # Clean up common prefixes
            domain = domain.replace('mail.', '').replace('noreply.', '')
            return domain
        return None

import requests
from bs4 import BeautifulSoup
import re

class ContactFinder:
    def __init__(self):
        self.common_paths = [
            '/privacy', '/privacy-policy', '/legal/privacy',
            '/privacy-center', '/legal', '/terms'
        ]
        self.email_patterns = [
            r'privacy@[a-zA-Z0-9.-]+',
            r'dpo@[a-zA-Z0-9.-]+',
            r'data-protection@[a-zA-Z0-9.-]+',
            r'gdpr@[a-zA-Z0-9.-]+'
        ]
        
    def find_contact(self, domain):
        """Attempt to find privacy contact automatically"""
        base_url = f"https://{domain}"
        
        # Try common privacy page URLs
        for path in self.common_paths:
            try:
                url = base_url + path
                response = requests.get(url, timeout=5, allow_redirects=True)
                
                if response.status_code == 200:
                    contact = self.extract_contact(response.text, domain)
                    if contact:
                        return contact
                        
            except Exception:
                continue
                
        # Try common email patterns
        for pattern in ['privacy', 'dpo', 'gdpr', 'data-protection']:
            test_email = f"{pattern}@{domain}"
            if self.verify_email_format(test_email):
                return {'type': 'email', 'contact': test_email, 'confidence': 'low'}
                
        return None
        
    def extract_contact(self, html_content, domain):
        """Extract email from HTML content"""
        soup = BeautifulSoup(html_content, 'html.parser')
        text = soup.get_text()
        
        # Look for email addresses
        for pattern in self.email_patterns:
            matches = re.findall(pattern, text, re.IGNORECASE)
            if matches:
                return {
                    'type': 'email',
                    'contact': matches[0],
                    'confidence': 'high'
                }
                
        # Look for contact forms
        forms = soup.find_all('form')
        for form in forms:
            if any(word in str(form).lower() for word in ['privacy', 'deletion', 'gdpr']):
                return {
                    'type': 'webform',
                    'contact': form.get('action', 'Found webform'),
                    'confidence': 'medium'
                }
                
        return None

import base64
from email.mime.text import MIMEText

class RequestSender:
    def __init__(self, gmail_service):
        self.service = gmail_service
        self.templates = DeletionTemplates()
        
    def send_deletion_request(self, company, account_email, template_type='generic'):
        """Send deletion request email"""
        # Get template
        template_func = self.templates.templates[template_type]
        email_content = template_func(
            company['company_name'],
            account_email
        )
        
        # Create message
        message = MIMEText(email_content['body'])
        message['to'] = company['privacy_contact']
        message['from'] = account_email
        message['subject'] = email_content['subject']
        
        # Encode and send
        raw_message = base64.urlsafe_b64encode(
            message.as_bytes()
        ).decode('utf-8')
        
        try:
            sent_message = self.service.users().messages().send(
                userId='me',
                body={'raw': raw_message}
            ).execute()
            
            return {
                'success': True,
                'message_id': sent_message['id'],
                'thread_id': sent_message.get('threadId')
            }
            
        except Exception as e:
            return {
                'success': False,
                'error': str(e)
            }

class ReplyTracker:
    def __init__(self, gmail_service):
        self.service = gmail_service
        self.db_path = Path.home() / '.data_deletion_tool' / 'local.db'
        
    def check_for_replies(self):
        """Check all pending requests for replies"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # Get all sent requests
        cursor.execute('''
            SELECT r.id, r.sent_message_id, c.company_name
            FROM requests r
            JOIN companies c ON r.company_id = c.id
            WHERE r.status = 'SENT' 
            AND r.sent_message_id IS NOT NULL
        ''')
        
        pending_requests = cursor.fetchall()
        
        for request_id, message_id, company_name in pending_requests:
            # Get the thread
            try:
                thread = self.service.users().threads().get(
                    userId='me',
                    id=message_id
                ).execute()
                
                # Check if there are replies (more than 1 message in thread)
                if len(thread['messages']) > 1:
                    self.process_reply(request_id, thread['messages'][-1])
                    print(f"✓ Reply received from {company_name}")
                    
            except Exception as e:
                print(f"Error checking {company_name}: {e}")
                
        conn.close()
        
    def process_reply(self, request_id, reply_message):
        """Analyze reply and update status"""
        # Get reply content
        snippet = reply_message.get('snippet', '').lower()
        
        # Categorize response
        if any(word in snippet for word in ['confirmed', 'deleted', 'completed', 'processed']):
            status = 'COMPLETED'
        elif any(word in snippet for word in ['processing', 'received', 'working']):
            status = 'REPLIED'
        else:
            status = 'REPLIED'
            
        # Update database
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute('''
            UPDATE requests 
            SET status = ?, reply_received_date = CURRENT_TIMESTAMP
            WHERE id = ?
        ''', (status, request_id))
        conn.commit()
        conn.close()
